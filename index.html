import streamlit as st import yfinance as yf import pandas as pd import numpy as np from datetime import date import logging # إعداد سجل الأحداث logging.basicConfig(level=logging.INFO, format='%(asctime)s - %(levelname)s - %(message)s') # ==================================================== # الفئة TechnicalAnalyzer: لتحليل بيانات الأسهم واكتشاف اختراق الشموع البيعية # ==================================================== class TechnicalAnalyzer: def __init__(self, lose_body_percent=0.55): self.lose_body_percent = lose_body_percent def detect_sell_breakout(self, df_symbol): n = len(df_symbol) o = df_symbol['Open'].values h = df_symbol['High'].values l = df_symbol['Low'].values c = df_symbol['Close'].values body_ratio = np.where(h != l, np.abs(o - c) / (h - l), 0.0) valid_sell_candle = (c < o) & (body_ratio >= self.lose_body_percent) valid_sell_high = np.empty(n, dtype=float) valid_sell_high[0] = np.nan breakout = np.full(n, False, dtype=bool) for i in range(1, n): prev_high = valid_sell_high[i - 1] if (not np.isnan(prev_high)) and (c[i] > prev_high) and (not valid_sell_candle[i]): breakout[i] = True valid_sell_high[i] = np.nan else: breakout[i] = False if valid_sell_candle[i]: valid_sell_high[i] = h[i] else: valid_sell_high[i] = prev_high df_symbol = df_symbol.copy() df_symbol['validSellCandle'] = valid_sell_candle df_symbol['validSellHigh'] = valid_sell_high df_symbol['breakout'] = breakout return df_symbol # ==================================================== # واجهة المستخدم باستخدام Streamlit # ==================================================== st.set_page_config(page_title="تحليل اختراق الشموع البيعية", layout="wide") st.title("📊 تطبيق تحليل اختراق الشموع البيعية") st.markdown("""أدخل رموز الأسهم (مثل: AAPL, MSFT) واختر الفاصل الزمني، وسنقوم بتحليل البيانات واكتشاف الاختراقات للشموع البيعية بناءً على النسبة المحددة.""") symbols_input = st.text_area("🔤 الرموز (افصلها بفاصلة أو مسافة):", value="AAPL MSFT GOOGL") interval = st.selectbox("⏱️ اختر الفاصل الزمني:", ["1d", "1wk", "1mo"]) start_date = st.date_input("📅 تاريخ البداية:", value=pd.to_datetime("2000-01-01")) end_date = st.date_input("📅 تاريخ النهاية:", value=date.today()) if st.button("🚀 تنفيذ التحليل"): symbols = [sym.strip().upper() for sym in symbols_input.replace(",", " ").split() if sym.strip()] try: st.info("🔄 يتم تحميل البيانات... الرجاء الانتظار") data = yf.download( tickers=symbols, start=start_date, end=end_date, interval=interval, group_by="ticker", auto_adjust=True, progress=False, threads=True ) st.success("✅ تم تحميل البيانات بنجاح") analyzer = TechnicalAnalyzer(lose_body_percent=0.55) breakout_results = [] for symbol in symbols: if symbol in data: df_symbol = data[symbol].reset_index() result_df = analyzer.detect_sell_breakout(df_symbol) last_row = result_df.tail(1) if last_row['breakout'].iloc[0]: breakout_results.append((symbol, round(last_row['Close'].iloc[0], 2))) else: st.warning(f"⚠️ لم يتم العثور على بيانات للرمز: {symbol}") if breakout_results: st.subheader("📈 النتائج:") result_df = pd.DataFrame(breakout_results, columns=["الرمز", "سعر الإغلاق"]) st.dataframe(result_df) else: st.info("ℹ️ لا توجد رموز تحقق شرط الاختراق في آخر شمعة.") except Exception as e: st.error(f"❌ حدث خطأ أثناء جلب البيانات أو التحليل: {e}")
